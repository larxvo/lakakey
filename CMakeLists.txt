cmake_minimum_required(VERSION 3.16)
project(Lakakey VERSION 0.7.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LAKAKEY_BUILD_TESTS "Build engine tests" ON)
add_library(lakakey_core STATIC src/engine.cpp)
target_include_directories(lakakey_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(lakakey_core PRIVATE
    $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic>
    $<$<CXX_COMPILER_ID:MSVC>:/W4 /utf-8>)
set_target_properties(lakakey_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(lakakey_c SHARED src/c_api.cpp)
target_link_libraries(lakakey_c PRIVATE lakakey_core)
target_include_directories(lakakey_c PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(lakakey_c PRIVATE LAKAKEY_BUILD_DLL)
add_executable(lakakey_c_example examples/c_client.c)
target_include_directories(lakakey_c_example PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(lakakey_c_example PRIVATE lakakey_c)
add_executable(lakakey_lab src/lab.cpp)
target_link_libraries(lakakey_lab PRIVATE lakakey_core)
if(LAKAKEY_BUILD_TESTS)
    enable_testing()
    add_executable(lakakey_benchmark tests/benchmark.cpp)
    target_link_libraries(lakakey_benchmark PRIVATE lakakey_core)
    add_executable(lakakey_tests tests/test_engine.cpp)
    target_link_libraries(lakakey_tests PRIVATE lakakey_core)
    add_test(NAME lakakey_engine_tests COMMAND lakakey_tests)
    add_executable(lakakey_session_tests tests/test_session.cpp)
    target_link_libraries(lakakey_session_tests PRIVATE lakakey_core)
    add_test(NAME lakakey_session_tests COMMAND lakakey_session_tests)
endif()

# Windows-native settings companion (not a system IME).
if(WIN32)
  add_executable(LakakeySettings WIN32 platform/windows/settings.cpp platform/windows/version.rc)
  target_compile_definitions(LakakeySettings PRIVATE UNICODE _UNICODE)
  target_link_libraries(LakakeySettings PRIVATE shell32 ole32)
  add_executable(LakakeyMascot WIN32 platform/windows/mascot/runtime.cpp platform/windows/mascot/version.rc)
  target_compile_definitions(LakakeyMascot PRIVATE UNICODE _UNICODE)
  target_link_libraries(LakakeyMascot PRIVATE gdiplus shell32 user32 gdi32)
  add_executable(LakakeyMascotEvent platform/windows/mascot/send_event.cpp)
  target_compile_definitions(LakakeyMascotEvent PRIVATE UNICODE _UNICODE)
  add_custom_command(TARGET LakakeyMascot POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_CURRENT_SOURCE_DIR}/platform/windows/mascot/assets
    $<TARGET_FILE_DIR:LakakeyMascot>/assets)
  if(MSVC)
    target_compile_options(LakakeySettings PRIVATE /W4 /utf-8)
  endif()
endif()

  target_link_libraries(LakakeyTsfProbe PRIVATE ole32 msctf)
endif()

# Experimental Windows TSF COM DLL: registration & key sink, not yet an operational IME.
option(LAKAKEY_BUILD_EXPERIMENTAL_TSF "Build unverified experimental TSF COM DLL (not for end users)" OFF)

  target_include_directories(LakakeyTsf PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_definitions(LakakeyTsf PRIVATE UNICODE _UNICODE)
endif()
