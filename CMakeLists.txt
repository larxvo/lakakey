cmake_minimum_required(VERSION 3.16)
project(Lakakey VERSION 0.14.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(LAKAKEY_BUILD_TESTS "Build engine tests" ON)
add_library(lakakey_core STATIC src/engine.cpp)
target_include_directories(lakakey_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(lakakey_core PRIVATE
    $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall -Wextra -Wpedantic>
    $<$<CXX_COMPILER_ID:MSVC>:/W4 /utf-8>)
set_target_properties(lakakey_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
add_library(lakakey_c SHARED src/c_api.cpp)
target_link_libraries(lakakey_c PRIVATE lakakey_core)
target_include_directories(lakakey_c PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_definitions(lakakey_c PRIVATE LAKAKEY_BUILD_DLL)
add_executable(lakakey_c_example examples/c_client.c)
target_include_directories(lakakey_c_example PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(lakakey_c_example PRIVATE lakakey_c)
add_executable(lakakey_lab src/lab.cpp)
target_link_libraries(lakakey_lab PRIVATE lakakey_core)
if(LAKAKEY_BUILD_TESTS)
    enable_testing()
    add_executable(lakakey_benchmark tests/benchmark.cpp)
    target_link_libraries(lakakey_benchmark PRIVATE lakakey_core)
    add_executable(lakakey_tests tests/test_engine.cpp)
    target_link_libraries(lakakey_tests PRIVATE lakakey_core)
    add_test(NAME lakakey_engine_tests COMMAND lakakey_tests)
    add_executable(lakakey_session_tests tests/test_session.cpp)
    target_link_libraries(lakakey_session_tests PRIVATE lakakey_core)
    add_test(NAME lakakey_session_tests COMMAND lakakey_session_tests)
    add_executable(lakakey_contract_tests tests/test_contract.cpp)
    target_link_libraries(lakakey_contract_tests PRIVATE lakakey_core)
    if(MSVC)
      target_compile_options(lakakey_contract_tests PRIVATE /W4 /utf-8)
    endif()
    add_test(NAME lakakey_contract_tests COMMAND lakakey_contract_tests)
    add_executable(lakakey_tsf_policy_tests tests/test_tsf_policy.cpp)
    target_include_directories(lakakey_tsf_policy_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
    if(MSVC)
      target_compile_options(lakakey_tsf_policy_tests PRIVATE /W4 /utf-8)
    endif()
    add_test(NAME lakakey_tsf_policy_tests COMMAND lakakey_tsf_policy_tests)
    # Older CI jobs build only lakakey_contract_tests before running all CTest cases.
    add_dependencies(lakakey_contract_tests lakakey_tsf_policy_tests)
endif()

# Windows-native settings companion (not a system IME).
if(WIN32)
  add_executable(LakakeySettings WIN32 platform/windows/settings.cpp platform/windows/version.rc)
  target_compile_definitions(LakakeySettings PRIVATE UNICODE _UNICODE)
  target_link_libraries(LakakeySettings PRIVATE shell32 ole32)
  add_executable(LakakeyMascot WIN32 platform/windows/mascot/runtime.cpp platform/windows/mascot/version.rc)
  target_compile_definitions(LakakeyMascot PRIVATE UNICODE _UNICODE)
  target_link_libraries(LakakeyMascot PRIVATE gdiplus shell32 user32 gdi32)
  add_executable(LakakeyMascotEvent platform/windows/mascot/send_event.cpp)
  target_compile_definitions(LakakeyMascotEvent PRIVATE UNICODE _UNICODE)
  add_custom_command(TARGET LakakeyMascot POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
    ${CMAKE_CURRENT_SOURCE_DIR}/platform/windows/mascot/assets
    $<TARGET_FILE_DIR:LakakeyMascot>/assets)
  if(MSVC)
    target_compile_options(LakakeySettings PRIVATE /W4 /utf-8)
  endif()
endif()

# TSF availability probe, not a registered IME / text service.
if(WIN32)
  add_executable(LakakeyTsfProbe platform/windows/tsf/tsf_probe.cpp)
  target_link_libraries(LakakeyTsfProbe PRIVATE ole32 uuid)
endif()

# Experimental Windows TSF COM DLL: registration & key sink, not yet an operational IME.
option(LAKAKEY_BUILD_EXPERIMENTAL_TSF "Build unverified experimental TSF COM DLL (not for end users)" OFF)
if(WIN32 AND LAKAKEY_BUILD_EXPERIMENTAL_TSF)
  add_library(LakakeyTsf SHARED platform/windows/tsf/lakakey_tsf.cpp platform/windows/tsf/lakakey_tsf.def)
  target_link_libraries(LakakeyTsf PRIVATE lakakey_core ole32 uuid advapi32 user32)
  target_include_directories(LakakeyTsf PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_definitions(LakakeyTsf PRIVATE UNICODE _UNICODE)
endif()

# Opt-in TSF v2 native composition prototype. Never packaged into preview setup.
option(LAKAKEY_BUILD_TSF_V2 "Build experimental native TSF composition v2" OFF)
if(WIN32 AND LAKAKEY_BUILD_TSF_V2)
  add_library(LakakeyTsfV2 SHARED
    platform/windows/tsf/lakakey_tsf_v2.cpp
    platform/windows/tsf/lakakey_tsf_v2.def)
  target_link_libraries(LakakeyTsfV2 PRIVATE lakakey_core ole32 uuid advapi32 user32)
  target_include_directories(LakakeyTsfV2 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_compile_definitions(LakakeyTsfV2 PRIVATE UNICODE _UNICODE)
  set_target_properties(LakakeyTsfV2 PROPERTIES OUTPUT_NAME LakakeyTsfV2)
  # Explicitly invoked only in disposable Windows CI; never ordinary CTest.
  add_executable(LakakeyTsfRegistrationSmoke tests/windows/tsf_registration_smoke.cpp)
  target_link_libraries(LakakeyTsfRegistrationSmoke PRIVATE advapi32)
  if(MSVC)
    target_compile_options(LakakeyTsfRegistrationSmoke PRIVATE /W4 /utf-8)
  endif()
  # Verify physical-key translation against an actual Windows keyboard layout.
  add_executable(LakakeyKeyboardLayoutSmoke tests/windows/keyboard_layout_smoke.cpp)
  target_include_directories(LakakeyKeyboardLayoutSmoke PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  target_link_libraries(LakakeyKeyboardLayoutSmoke PRIVATE user32)
  if(MSVC)
    target_compile_options(LakakeyKeyboardLayoutSmoke PRIVATE /W4 /utf-8)
  endif()
  if(LAKAKEY_BUILD_TESTS)
    add_test(NAME lakakey_keyboard_layout_smoke COMMAND LakakeyKeyboardLayoutSmoke)
    add_dependencies(lakakey_contract_tests LakakeyKeyboardLayoutSmoke)
  endif()
  # Isolated COM probe: loads the DLL but NEVER registers or activates it.
  add_executable(LakakeyTsfComSmoke tests/windows/tsf_com_smoke.cpp)
  target_compile_definitions(LakakeyTsfComSmoke PRIVATE UNICODE _UNICODE)
  target_link_libraries(LakakeyTsfComSmoke PRIVATE ole32 uuid)
  if(MSVC)
    target_compile_options(LakakeyTsfComSmoke PRIVATE /W4 /utf-8)
  endif()
  if(LAKAKEY_BUILD_TESTS)
    add_test(NAME lakakey_tsf_com_smoke
             COMMAND $<TARGET_FILE:LakakeyTsfComSmoke> $<TARGET_FILE:LakakeyTsfV2>)
    # The legacy Windows job builds core tests by explicit target name.
    add_dependencies(lakakey_contract_tests LakakeyTsfComSmoke)
  endif()

  # Verify the text-service event sink against a real Windows TSF thread manager.
  # Does not register an input profile or capture text.
  add_executable(LakakeyTsfActivationSmoke tests/windows/tsf_activation_smoke.cpp)
  target_link_libraries(LakakeyTsfActivationSmoke PRIVATE ole32 uuid user32)
  if(MSVC)
    target_compile_options(LakakeyTsfActivationSmoke PRIVATE /W4 /utf-8)
  endif()
  if(LAKAKEY_BUILD_TESTS)
    add_test(NAME lakakey_tsf_activation_smoke
             COMMAND $<TARGET_FILE:LakakeyTsfActivationSmoke> $<TARGET_FILE:LakakeyTsfV2>)
    add_dependencies(lakakey_contract_tests LakakeyTsfActivationSmoke)
  endif()
  # Interactive synthetic typing probe, allowed ONLY on disposable CI runners.
  # Not part of CTest or the ordinary user installer; results are reported separately.
  add_executable(LakakeyTsfLiveTyping tests/windows/tsf_live_typing.cpp)
  target_link_libraries(LakakeyTsfLiveTyping PRIVATE ole32 uuid user32 advapi32)
  if(MSVC)
    target_compile_options(LakakeyTsfLiveTyping PRIVATE /W4 /utf-8)
  endif()
  # QA only: built with opt-in TSF research target, never in user installer.
  add_executable(LakakeyTsfSmokeHost WIN32 platform/windows/tsf/smoke_host.cpp)
  target_compile_definitions(LakakeyTsfSmokeHost PRIVATE UNICODE _UNICODE)
  target_link_libraries(LakakeyTsfSmokeHost PRIVATE user32 gdi32)
  if(MSVC)
    target_compile_options(LakakeyTsfV2 PRIVATE /W4 /utf-8)
  endif()
endif()
